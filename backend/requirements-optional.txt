# Forecasters kept out of requirements.txt, so a plain checkout stays light.
# The engine detects what is present and reports the rest as unavailable
# rather than failing (see app/forecasting/models.py: unavailable_models).
#
#   pip install -r requirements-optional.txt
#
# Deployed images install this too — see the INSTALL_OPTIONAL_MODELS build arg
# in the Dockerfile, which defaults to true.
prophet==1.1.6

# Pinned, and the upper bound is the point. Prophet's wheel bundles a trimmed
# CmdStan tree — `prophet/stan_model/cmdstan-2.33.1/` holds `bin/` and `stan/`
# and no makefile, because the model binary next to it is already compiled and
# nothing here ever builds one. cmdstanpy 1.3.0 made its path validation
# insist on that makefile, so it rejects the very tree Prophet ships:
#
#   ValueError: CmdStan installataion missing makefile, path
#   .../prophet/stan_model/cmdstan-2.33.1 is invalid
#
# Prophet asks only for `cmdstanpy>=1.0.4`, so an unpinned install resolves to
# 1.3.0 and breaks. It breaks quietly, too: `import prophet` still succeeds, so
# availability checks pass and the failure only lands at fit time. 1.2.5 is the
# last release that accepts the bundled tree.
cmdstanpy==1.2.5
